5. Números Naturales: Descomposición y Posición K
Regla Garzón: Las posiciones se cuentan desde la DERECHA (Pos 1 = unidades, Pos 2 = decenas...). 
function EliminarDigitoPosK(n: LongWord; k: Byte): LongWord;
var res, pot: LongWord; pos, d: Byte;
begin
  res := 0; pot := 1; pos := 1;
  while n > 0 do begin
    d := n mod 10;
    if pos <> k then begin
      res := res + (d * pot);
      pot := pot * 10;
    end;
    pos := pos + 1;
    n := n div 10;
  end;
  Result := res;
end;
6. Series Numéricas: Sin Math ni Power
Si el examen dice "No usar potencia", hazlo con bucle while: 
function Potencia(base, exp: Integer): Int64;
var p: Int64; c: Integer;
begin
  p := 1; c := 1;
  while c <= exp do begin
    p := p * base; c := c + 1;
  end;
  Result := p;
end;

// Denominador Fibonacci Aditivo (Semillas a=1, b=4 → 1, 4, 5, 9, 14...)
function DenomFib(k: Integer): Int64;
var a, b, c: Int64; i: Integer;
begin
  if k = 1 then Result := 1
  else if k = 2 then Result := 4
  else begin
    a := 1; b := 4; c := 5; i := 3;
    while i <= k do begin
      c := a + b; a := b; b := c; i := i + 1;
    end;
    Result := c;
  end;
end;
7. Factorial y Fibonacci Básico
function Factorial(n: Integer): Int64;
var f: Int64; c: Integer;
begin
  f := 1; c := 1;
  while c <= n do begin f := f * c; c := c + 1; end;
  Result := f;
end;

8. Metodología Oficial: Recorrido Palabra por Palabra
Usa punteros pIni y pFin para aislar cada palabra sin perder espacios: 
function MayusculasSiIniciaVocal(cad: String): String;
var n, i, pIni, pFin: Byte; pal, res: String;
begin
  n := Length(cad); res := ''; i := 1;
  while i <= n do begin
    if cad[i] = ' ' then begin
      res := res + ' '; i := i + 1;
    end else begin
      pIni := i;
      while (i <= n) and (cad[i] <> ' ') do i := i + 1;
      pFin := i - 1;
      pal := Copy(cad, pIni, pFin - pIni + 1);
      // Condición del examen: si inicia con vocal:
      if (Length(pal) > 0) and EsVocal(pal[1]) then
        pal := UpperCase(pal);
      res := res + pal;
    end;
  end;
  Result := res;
end;
9. Ordenación Burbuja en Cadenas (Pizarra Docente)
Doble bucle: I desde 1 hasta N-1, J desde I+1 hasta N: 
function OrdenarBurbujaAsc(pal: String): String;
var n, i, j: Byte; aux: Char;
begin
  n := Length(pal); i := 1;
  while i <= n - 1 do begin
    j := i + 1;
    while j <= n do begin
      if pal[i] > pal[j] then begin
        aux := pal[i];
        pal[i] := pal[j];
        pal[j] := aux;
      end;
      j := j + 1;
    end;
    i := i + 1;
  end;
  Result := pal;
end;
10. Invertir Última Palabra (Examen 2024-1 Z2)
function InvertirUltimaPalabra(cad: String): String;
var n, i, finPal, iniPal, k: Byte; palInv: String;
begin
  n := Length(cad); i := n;
  while (i >= 1) and (cad[i] = ' ') do i := i - 1;
  finPal := i;
  while (i >= 1) and (cad[i] <> ' ') do i := i - 1;
  iniPal := i + 1;
  palInv := ''; k := finPal;
  while k >= iniPal do begin
    palInv := palInv + cad[k]; k := k - 1;
  end;
  Result := Copy(cad, 1, iniPal - 1) + palInv + Copy(cad, finPal + 1, n - finPal);
end;

12. Eliminar Consonantes (Examen 2024-1 Z2)
function EliminarConsonantes(cad: String): String;
var res: String; i, n: Byte;
begin
  res := ''; n := Length(cad); i := 1;
  while i <= n do begin
    // Conserva vocales, espacios y puntuación
    if EsVocal(cad[i]) or not (UpCase(cad[i]) in ['A'..'Z']) then
      res := res + cad[i];
    i := i + 1;
  end;
  Result := res;
end;
13. ⚠️ Las 5 Trampas que Quitan Puntos en el Examen
  1. No inicializar acumuladores: Si no pones suma := 0; o res := ''; tu código acumula basura de memoria.
  2. Bucle infinito en while: Olvidar i := i + 1; o n := n div 10; cuelga el programa.
  3. Confundir función y procedimiento: La función usa Result := ...;. El procedimiento usa parámetros var y NO lleva Result.
  4. Posición K al revés: Recuerda que en números naturales, las unidades son la posición 1 (desde la derecha).
  5. Olvidar la fila final en la traza: La fila donde i > n (condición Falsa) es obligatoria para la docente.
